SECURE-RADS-BUILD(a, tar)
Input:
  a  ← algorithm id
  tar ← algorithm’s Docker image packaged as TAR 
1  cost dynamodb:read                   // registry getAlgorithm (existing key)
2  emit build-started
3  oci ← CONVERT-TO-OCI(tar)             // skopeo
4  rootfs ← UNPACK(oci)                 // umoci
5  LAYER sce-unit, nsjail onto rootfs
6  ⟨EIF, PCRs⟩ ← EIF-BUILD(rootfs)
7  emit build-completed
8  UPLOAD EIF to S3
9  cost s3:put                          // multipart initiation, parts, and completion
10 emit image-uploaded
11 REGISTER-PCRS(a, PCRs)               // mTLS to Verifier (verifier API GW/Lambda metered)
12 emit pcrs-registered
13 kmsArn ← CREATE-OR-REUSE-KMS-KEY(a)  // KMS excluded from cost
14 emit kms-key-provisioned
15 write a ↦ ⟨S3-path, kmsArn⟩ to DynamoDB
16 cost dynamodb:write
17 emit algorithm-registered
18 return


SECURE-RADS-RUN(a, n)
Input:
  a ← algorithm id
  n ← dataset size (# of records)
1  runId ← NEW-ID()
2  emit run-requested
3  cost dynamodb:read                   // execution-service: algorithm lookup
4  key ← LOOKUP-EIF(a) // S3 location of algorithm's EIF
5  RUN-COUNT-CHECK(n)
6  cost dynamodb:read-capacity          // PDM Query count, actual consumed capacity
7  RunInstances(Nitro parent, userdata ← ⟨a, runId, n, key, launchEpoch⟩)
8  emit executor-launched
9  cost apigateway:request + lambda     // per control-plane request/poll (GET /run/{runId} too)
10 return runId


SECURE-RADS-EXECUTOR(a, runId, n, key)
Input:
  a ← algorithm id
  runId ← run id
  n ← dataset size (# of records)
  key ←  S3 location of algorithm's EIF
1  install nitro-cli
2  configure allocator
3  emit executor-ready
4  cost s3:get                          // EIF pull
5  EIF ← GET-S3(key)
6  emit artifact-retrieved
7  start vsock listeners                // output :5000, runInfo :5001
8  configure vsock-proxy
9  RUN-ENCLAVE(EIF)
10 emit enclave-launched
11 wait for enclave exit
12 publish result to SQS
13 cost sqs:send                        // message bytes
14 emit executor-finished
15 terminate instance
16 cost ec2:instance                    // seconds from launchEpoch to this terminate call


SCE-UNIT(a)
Input:
  a ← algorithm id
1  cfg ← LOAD-RUN-CONFIG()
2  ⟨runId, n⟩ ← READ-RUNINFO(:5001)
3  emit enclave-started
4  SEED-ENTROPY from NSM                 // no AWS cost
5  emit unit-ready
6  token ← ATTEST(Verifier, runId)      // verifier API GW/Lambda metered
7  emit attested
8  creds ← FETCH-AWS-CREDS()            // host IMDS proxy, no AWS cost
9  ⟨records, encDEK⟩ ← FETCH-PDM(a, token, limit ← n)
10 cost dynamodb:read                   // emitted by pdm-service: data-key GetItem + records Query pages
11 emit records-fetched
12 DEK ← KMS-DECRYPT(encDEK, attestation)   // KMS excluded from cost
13 plaintext ← DECRYPT(records, DEK)
14 emit records-decrypted
15 write {algorithmId, records} to /data/records.json
16 emit records-stored-locally
17 START nsjail                         // RECORDS_PATH, RESULT_PATH
18 wait for nsjail exit
19 output ← read /data/result.json
20 emit dc-app-completed
21 STREAM output to parent              // vsock :5000
22 emit output-sent
23 // on failure: emit run-failed


SECURE-RADS-RESULT-CONSUMER(result)
Input:
  result ← processing result returned by the SCE-UNIT
1  cost sqs:receive + dynamodb:write    // result-consumer batch
2  persist result to DynamoDB
3  emit result-persisted

SECURE-RADS-BUILD(a, oci)
1  buildId ← NEW-ID()
2  emit build-started
3  rootfs ← UNPACK(oci)                 // skopeo + umoci
4  LAYER sce-unit, nsjail onto rootfs
5  ⟨EIF, PCRs⟩ ← EIF-BUILD(rootfs)
6  emit eif-built
7  UPLOAD EIF to S3
8  emit image-uploaded
9  REGISTER-PCRS(a, PCRs)               // mTLS to Verifier
10 emit pcrs-registered
11 kmsArn ← CREATE-OR-REUSE-KMS-KEY(a)
12 emit kms-key-provisioned
13 write a ↦ ⟨S3-path, kmsArn⟩ to DynamoDB
14 emit build-completed                 // or build-failed
15 return


SECURE-RADS-RUN(a, n)
1  runId ← NEW-ID()
2  emit run-requested
3  key ← LOOKUP-EIF(a)
4  RunInstances(Nitro parent, userdata ← ⟨a, runId, n, key⟩)
5  emit enclave-launched
6  return runId


SECURE-RADS-HOST(a, runId, n, key)
1  install nitro-cli
2  configure allocator
3  emit executor-ready
4  pull EIF from S3
5  emit artifact-retrieved
6  start vsock listeners                // output :5000, runInfo :5001
7  RUN-ENCLAVE(EIF)
8  wait for enclave exit
9  publish result to SQS
10 terminate instance


SCE-UNIT(a, cfg)
1  ⟨runId, n⟩ ← READ-RUNINFO(:5001)
2  emit enclave-started
3  SEED-ENTROPY from NSM
4  emit unit-ready
5  token ← ATTEST(Verifier)
6  emit attested
7  creds ← FETCH-AWS-CREDS()
8  ⟨records, encDEK, pages⟩ ← FETCH-PDM(a, token, limit ← n)
9  emit records-fetched
10 DEK ← KMS-DECRYPT(encDEK, attestation)
11 plaintext ← DECRYPT(records, DEK)
12 emit data-decrypted
13 write {algorithmId, pages, records} to /data/records.json
14 START nsjail                         // RECORDS_PATH, RESULT_PATH
15 wait for nsjail exit
16 output ← read /data/result.json
17 emit dc-app-completed
18 STREAM output to parent              // vsock :5000
19 emit output-sent
20 // on failure: emit run-failed


SECURE-RADS-RESULT-CONSUMER(result)
1  persist result to DynamoDB
2  emit result-persisted

SECURE-RADS-BUILD(a, tar)
Input:
  a  ← algorithm id
  tar ← algorithm’s Docker image packaged as TAR 
1  cost dynamodb:read-capacity          // registry getAlgorithm (existing key)
2  emit build-started
3  oci ← CONVERT-TO-OCI(tar)             // skopeo
4  rootfs ← UNPACK(oci)                 // umoci
5  LAYER sce-unit, nsjail onto rootfs
6  ⟨EIF, PCRs⟩ ← EIF-BUILD(rootfs)
7  emit build-completed
8  UPLOAD EIF to S3
9  cost s3:put                          // multipart initiation, parts, and completion
10 emit image-uploaded
11 REGISTER-PCRS(a, PCRs)               // verifier API GW/Lambda and DynamoDB write metered
12 emit pcrs-registered
13 kmsArn ← CREATE-OR-REUSE-KMS-KEY(a)  // policy names the kms-service role, gated on PCRs; KMS excluded from cost
14 emit kms-key-provisioned
15 write a ↦ ⟨S3-path, kmsArn⟩ to DynamoDB
16 cost dynamodb:write-capacity
17 emit algorithm-registered
18 return


SECURE-RADS-RUN(a, n)
Input:
  a ← algorithm id
  n ← dataset size (# of records)
1  runId ← NEW-ID()
2  emit run-requested
3  cost dynamodb:read-capacity          // execution-service: algorithm lookup
4  key ← LOOKUP-EIF(a) // S3 location of algorithm's EIF
5  RUN-COUNT-CHECK(n)
6  cost dynamodb:read-capacity          // execution-service Query on the PDM table
7  RunInstances(Nitro parent, userdata ← ⟨a, runId, n, key, launchEpoch⟩)
8  emit executor-launched
9  cost apigateway:request + lambda     // per control-plane request/poll (GET /run/{runId}?algorithmId=... too)
10 return runId


SECURE-RADS-EXECUTOR(a, runId, n, key)
Input:
  a ← algorithm id
  runId ← run id
  n ← dataset size (# of records)
  key ←  S3 location of algorithm's EIF
1  install nitro-cli
2  configure allocator
3  emit executor-ready                  // direct, auth none
4  cost s3:get                          // EIF pull
5  EIF ← GET-S3(key)
6  emit artifact-retrieved              // direct, auth none
7  start vsock listeners                // output :5000, runInfo :5001
8  configure vsock-proxy
9  RUN-ENCLAVE(EIF)
10 emit enclave-launched                 // direct, auth none
11 wait for enclave exit
12 publish result to SQS
13 cost sqs:send                        // message bytes
14 emit executor-finished               // direct, auth none
15 cost ec2:instance                    // captured before the cost POST and terminate request
16 terminate instance


SCE-UNIT(a)
Input:
  a ← algorithm id
1  cfg ← LOAD-RUN-CONFIG()
2  ⟨runId, n⟩ ← READ-RUNINFO(:5001)
3  emit enclave-started                 // no token yet, auth none
4  SEED-ENTROPY from NSM                 // no AWS cost
5  emit unit-ready                      // no token yet, auth none
6  START nsjail                         // SCE_RPC_SOCKET=/data/sce.sock, ALGORITHM_ID
7  wait for app FetchRecords RPC        // first call triggers the steps below; retries reuse the outcome
8  challenge ← REQUEST-CHALLENGE(Verifier, a, runId)   // single-use nonce, verifier API GW/Lambda and DynamoDB metered
9  token ← ATTEST(Verifier, runId, challenge)          // NSM nonce ← challenge; claims the run so only one attestation succeeds; verifier API GW/Lambda and DynamoDB metered
10 emit attested
11 records ← FETCH-PDM(a, token, limit ← n)
12 cost apigateway:request + lambda:invocation + lambda:compute + dynamodb:read-capacity // PDM /records
13 emit records-fetched
14 ciphertext ← KMS-SERVICE-DECRYPT(token)  // unwraps the data key under this enclave's measurements
15 cost apigateway:request + lambda:invocation + lambda:compute + dynamodb:read-capacity // kms-service /decrypt
16 DEK ← UNWRAP(ciphertext, enclaveKey)   // KMS request itself is excluded from cost
17 plaintext ← DECRYPT(records, DEK)      // RecordEnvelope -> sce.v1.Record
18 emit records-decrypted
19 emit records-stored-locally
20 STREAM records to app                 // gRPC FetchRecords, one sce.v1.Record per message
21 wait for app SubmitResult RPC         // google.protobuf.Struct
22 wait for nsjail exit
23 emit dc-app-completed               // on success, else run-failed
24 output ← JSON(result)
25 STREAM output to parent              // vsock :5000
26 emit output-sent
27 // on failure: emit run-failed



KMS-SERVICE-DECRYPT(token)
Input:
  token ← verifier access JWT (aud includes kms-service, sub is the algorithm id)
1  verify token against verifier JWKS          // aud, iss, exp, signature
2  a ← token.sub
3  ⟨s3-path, kmsArn⟩ ← GET-ALGORITHM(a)
4  wrapped ← GET-DATA-KEY(a)
5  cost dynamodb:read-capacity                 // algorithm and data-key GetItems
6  ⟨attestation⟩ ← READ-REQUEST()
7  ciphertext ← KMS-DECRYPT(kmsArn, wrapped, recipient ← attestation)
   // The key policy requires the attestation's PCR0/1/2 and KMS returns CiphertextForRecipient encrypted to
   // the enclave's ephemeral key: the service never sees plaintext, and only the measured enclave can unwrap.
8  return ciphertext to the enclave

SECURE-RADS-RESULT-CONSUMER(result)
Input:
  result ← processing result returned by the SCE-UNIT
1  SQS delivers result batch; invoke Lambda // receive is billed before the handler runs
2  persist result to DynamoDB
3  cost dynamodb:write-capacity; emit result-persisted
4  cost sqs:receive + lambda:invocation + lambda:compute // reported by handler after processing
5  cost sqs:delete                         // reported on success; Lambda deletes after return

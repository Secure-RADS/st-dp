RADS-BUILD(a, tar)
1  store tar in S3 under a
2  EMIT image-uploaded
3  write a ↦ S3-path to DynamoDB
4  EMIT algorithm-registered


RADS-RUN(a, n)
1  runId ← NEW-ID()
2  EMIT run-requested
3  key ← LOOKUP-S3(a)
4  RunInstances(userdata ← ⟨a, runId, n, key⟩)
5  EMIT executor-launched
6  return runId


RADS-EXECUTOR(a, runId, n, key)
1  INSTALL docker, awscli
2  START docker
3  EMIT executor-ready
4  PULL key from S3
5  DOCKER-LOAD tar
6  EMIT artifact-retrieved
7  records ← FETCH-PDM(a, limit ← n)
8  write records to records.json
9  EMIT records-fetched
10 DOCKER-RUN image with /data/records.json bind-mounted
11 EMIT compute-completed
12 PUBLISH result to SQS
13 EMIT executor-finished
14 TERMINATE instance


RADS-RESULT-CONSUMER(result)
1  PERSIST result to DynamoDB

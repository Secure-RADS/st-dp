RADS-BUILD(a, tar)
1  store tar in S3 under a
2  emit image-uploaded
3  write a ↦ S3-path to DynamoDB
4  emit algorithm-registered


RADS-RUN(a, n)
1  runId ← NEW-ID()
2  emit run-requested
3  key ← LOOKUP-S3(a)
4  RunInstances(userdata ← ⟨a, runId, n, key⟩)
5  emit executor-launched
6  return runId


RADS-EXECUTOR(a, runId, n, key)
1  install docker, awscli
2  start docker
3  pull rads-host-tools from S3
4  emit executor-ready
5  pull key from S3
6  DOCKER-LOAD(tar)
7  emit artifact-retrieved
8  records ← FETCH-PDM(a, limit ← n)   // rads-host-tools fetch-records
9  write records to records.json
10 emit records-fetched
11 DOCKER-RUN(image, bind ← /data/records.json)
12 emit compute-completed
13 publish result to SQS
14 emit executor-finished
15 terminate instance


RADS-RESULT-CONSUMER(result)
1  persist result to DynamoDB

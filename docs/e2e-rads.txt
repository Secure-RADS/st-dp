RADS-BUILD(a, tar)
Input:
  a  ← algorithm id
  tar ← algorithm’s Docker image packaged as a TAR 
1  UPLOAD tar to S3 
2  emit image-uploaded
3  write a ↦ S3-path to DynamoDB
4  emit algorithm-registered


RADS-RUN(a, n)
Input:
  a ← algorithm id
  n ← dataset size (# of records)
1  runId ← NEW-ID()
2  emit run-requested
3  key ← LOOKUP-S3(a) // S3 location of algorithm's TAR
4  RunInstances(userdata ← ⟨a, runId, n, key⟩)
5  emit executor-launched
6  return runId


RADS-EXECUTOR(a, runId, n, key)
Input:
  a ← algorithm id
  runId ← run id
  n ← dataset size (# of records)
  key ←  S3 location of algorithm's TAR
1  install docker, awscli
2  start docker
3  pull rads-host-tools from S3
4  emit executor-ready
5  tar ← GET-S3(key)
6  DOCKER-LOAD(tar)
7  emit artifact-retrieved
8  records ← FETCH-PDM(a, limit ← n)   // rads-host-tools fetch-records
9  emit records-fetched
10 write records to records.json
11 emit records-stored-locally
12 DOCKER-RUN(image, bind ← /data/records.json)
13 emit compute-completed
14 publish result to SQS
15 emit executor-finished
16 terminate instance


RADS-RESULT-CONSUMER(result)
Input:
  result ← processing result returned by the RADS executor
1  persist result to DynamoDB
2 emit result-persisted

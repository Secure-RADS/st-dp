RADS-BUILD(a, tar)
Input:
  a  ← algorithm id
  tar ← algorithm’s Docker image packaged as a TAR 
1  /upload: PRESIGN S3 PUT(a)           // presign is local, no S3 call
2  cost apigateway:request + lambda
3  client PUTs tar to S3
4  /register: HEAD a in S3
5  cost s3:head
6  cost s3:put                          // single presigned PutObject request
7  emit image-uploaded
8  write a ↦ ⟨bucket, key⟩ to DynamoDB
9  cost dynamodb:write-capacity
10 emit algorithm-registered
11 cost apigateway:request + lambda     // /register request
12 return


RADS-RUN(a, n)
Input:
  a ← algorithm id
  n ← dataset size (# of records)
1  runId ← NEW-ID()
2  emit run-requested
3  cost dynamodb:read-capacity          // execution-service: algorithm lookup
4  key ← LOOKUP-S3(a) // S3 location of algorithm's TAR
5  RUN-COUNT-CHECK(n)
6  cost dynamodb:read-capacity          // execution-service Query on the PDM table
7  RunInstances(userdata ← ⟨a, runId, n, key, launchEpoch⟩)
8  emit executor-launched
9  cost apigateway:request + lambda     // per control-plane request/poll (GET /run/{runId} too)
10 return runId


RADS-EXECUTOR(a, runId, n, key)
Input:
  a ← algorithm id
  runId ← run id
  n ← dataset size (# of records)
  key ←  S3 location of algorithm's TAR
1  install docker, awscli
2  start docker
3  pull rads-host-tools from S3
4  cost s3:get                          // rads-host-tools bundle, before the algorithm artifact
5  emit executor-ready
6  tar ← GET-S3(key)
7  DOCKER-LOAD(tar)
8  cost s3:get                          // algorithm artifact
9  emit artifact-retrieved
10 records ← FETCH-PDM(a, limit ← n)   // rads-host-tools fetch-records
11 cost apigateway:request + lambda:invocation + lambda:compute + dynamodb:read-capacity // PDM /records
12 emit records-fetched
13 write records to records.json
14 emit records-stored-locally
15 DOCKER-RUN(image, bind ← /data/records.json)
16 emit compute-completed
17 publish result to SQS
18 cost sqs:send                        // message bytes
19 emit executor-finished
20 cost ec2:instance                    // captured before the cost POST and terminate request
21 terminate instance


RADS-RESULT-CONSUMER(result)
Input:
  result ← processing result returned by the RADS executor
1  SQS delivers result batch; invoke Lambda // receive is billed before the handler runs
2  persist result to DynamoDB
3  cost dynamodb:write-capacity; emit result-persisted
4  cost sqs:receive + lambda:invocation + lambda:compute // reported by handler after processing
5  cost sqs:delete                         // reported on success; Lambda deletes after return
